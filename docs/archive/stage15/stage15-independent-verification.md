### Fresh Container Test (Docker)

```dockerfile
# Dockerfile.verification
FROM golang:1.27-alpine

# Install verification tools
RUN apk add --no-cache \
    cosign \
    jq \
    coreutils \
    git

WORKDIR /work

# Clone repository at exact tag
ARG TAG=v4.0.0-rc2
RUN git clone --depth 1 --branch ${TAG} https://github.com/Debajyoti0-0/aether.git .

# Run verification
RUN sha256sum -c checksums.txt

# For each Linux/macOS artifact
RUN for f in aether_${VERSION}_linux_*.tar.gz aether_${VERSION}_darwin_*.tar.gz; do \
    if [ -f "$f.sig" ] && [ -f "$f.pem" ]; then \
      cosign verify-blob --signature "$f.sig" --certificate "$f.pem" \
        --certificate-identity-regexp ".*" --certificate-oidc-issuer-regexp ".*" "$f"; \
    fi; \
  done

# Verify SBOM
RUN jq -e '.specVersion' sbom-cyclonedx.json

# Verify manifest
RUN jq -r '.artifacts[] | "\(.sha256)  \(.name)"' release-manifest.json | sha256sum -c
```
